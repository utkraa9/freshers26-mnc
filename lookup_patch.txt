async function lookupTicket(){
  const q=document.getElementById('lookupTicket').value.trim();
  const m=document.getElementById('lookupMsg');
  const box=document.getElementById('lookupQr');
  box.innerHTML='';
  document.getElementById('ticketCard').style.display='none';
  if(!q){m.textContent='Enter Roll Number, Reference ID, or Name.';m.className='sub err';return}
  m.textContent='Searching...';m.className='sub';
  try{
    const response=await fetch('/api/ticket-lookup?q='+encodeURIComponent(q),{headers:{Accept:'application/json'}});
    const result=await response.json().catch(()=>({}));
    if(!response.ok){
      m.textContent=response.status===404?'No registration found for that Roll Number, Reference ID, or Name.':'Verification service error. Please try again.';
      m.className='sub err';
      return;
    }
    document.getElementById('tid').textContent=result.ticket_id||'PENDING';
    document.getElementById('tn').textContent=result.name||'';
    document.getElementById('td').textContent=(result.roll_number||'')+' • '+String(result.category||'').toUpperCase();
    document.getElementById('ticketCard').style.display='block';
    if(result.payment_status==='verified'||result.ticket_status==='issued'){
      m.textContent='Pass verified! Digital clearance confirmed.';
      m.className='sub ok';
      document.getElementById('tstatus').textContent='CLEARED & VERIFIED';
      const token=result.ticket_qr_token||result.ticket_id;
      if(token)new QRCode(box,{text:token,width:170,height:170});
    }else{
      m.textContent='Registration found. Payment receipt is under verification with organizers.';
      m.className='sub';
      document.getElementById('tstatus').textContent='PAYMENT UNDER REVIEW';
    }
  }catch(error){
    console.error(error);
    m.textContent='Verification service error. Please try again.';
    m.className='sub err';
  }
}
